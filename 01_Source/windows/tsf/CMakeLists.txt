cmake_minimum_required(VERSION 3.21)
project(programming_ime_tsf LANGUAGES CXX)
if(NOT WIN32)
  message(FATAL_ERROR "Windows 11 SDK is required")
endif()
add_library(pime_tsf SHARED PimeTsf.cpp)
target_compile_features(pime_tsf PRIVATE cxx_std_17)
target_compile_definitions(pime_tsf PRIVATE UNICODE _UNICODE _WIN32_WINNT=0x0A00)
target_link_libraries(pime_tsf PRIVATE ole32 msctf advapi32)
set_target_properties(pime_tsf PROPERTIES OUTPUT_NAME "pime-tsf")
